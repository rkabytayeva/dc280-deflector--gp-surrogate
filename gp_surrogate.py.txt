import numpy as np
import matplotlib.pyplot as plt
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, WhiteKernel, ConstantKernel as C
import joblib
import time
from google.colab import files   # если вы в Colab

# ---------- Данные FLUKA (14 точек) ----------
time_h = np.array([0.5, 1, 2, 6, 12, 24, 36, 48, 72, 96, 120, 168, 200, 240])

dose_surf = np.array([77.0, 70.0, 63.0, 52.0, 45.0, 36.0, 31.0, 28.0, 24.0, 21.5, 20.0, 17.0, 15.5, 13.5])
dose_0_5_no = np.array([39.9, 36.0, 32.0, 27.0, 23.0, 19.0, 17.0, 15.5, 13.5, 12.0, 11.0, 10.0, 9.5, 9.0])
dose_0_5_pb = np.array([31.3, 28.5, 25.5, 22.0, 19.0, 16.0, 14.5, 13.0, 12.0, 11.0, 10.5, 10.0, 9.5, 9.2])

# ---------- Функция обучения GP на логарифме дозы ----------
def fit_gp_log(x, y):
    X = np.log10(x).reshape(-1, 1)
    log_y = np.log10(y)
    kernel = C(1.0, (0.1, 10.0)) * RBF(1.0, (0.2, 5.0)) + WhiteKernel(1.0)
    gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=10,
                                  alpha=1e-6, normalize_y=True)
    gp.fit(X, log_y)
    return gp

print("Training GP models...")
gp_surf = fit_gp_log(time_h, dose_surf)
gp_no   = fit_gp_log(time_h, dose_0_5_no)
gp_pb   = fit_gp_log(time_h, dose_0_5_pb)
print("Training completed.")

# ---------- Сохраняем модели в .pkl ----------
joblib.dump(gp_surf, 'gp_surface_new.pkl')
joblib.dump(gp_no, 'gp_50cm_no_new.pkl')
joblib.dump(gp_pb, 'gp_50cm_shield_new.pkl')
print("Models saved as .pkl files.")

# Скачиваем в Colab (если нужно)
try:
    files.download('gp_surface_new.pkl')
    files.download('gp_50cm_no_new.pkl')
    files.download('gp_50cm_shield_new.pkl')
    print("Download links generated.")
except:

# ---------- Построение графика ----------
tnew = np.logspace(np.log10(0.5), np.log10(300), 200)
Xnew = np.log10(tnew).reshape(-1, 1)

def predict_with_ci(gp, X, z=1.645):
    log_mean, log_std = gp.predict(X, return_std=True)
    mean = 10**log_mean
    lower = 10**(log_mean - z * log_std)
    upper = 10**(log_mean + z * log_std)
    return mean, lower, upper

fig, axs = plt.subplots(1, 3, figsize=(16, 5))
titles = ['Surface, mSv/h', '0.5 m, no shield, μSv/h', '0.5 m, 5 mm Pb, μSv/h']
data = [(dose_surf, gp_surf), (dose_0_5_no, gp_no), (dose_0_5_pb, gp_pb)]

for i, (ydata, gp) in enumerate(data):
    y_mean, y_low, y_high = predict_with_ci(gp, Xnew, z=1.645)  # 90% CI

    axs[i].fill_between(tnew, y_low, y_high, alpha=0.4, color='gray', label='90% CI')
    axs[i].plot(tnew, y_low, 'b--', linewidth=1.2, alpha=0.8)
    axs[i].plot(tnew, y_high, 'b--', linewidth=1.2, alpha=0.8, label='bounds')
    axs[i].plot(tnew, y_mean, 'b-', lw=2.5, label='GP mean')
    axs[i].scatter(time_h, ydata, color='r', s=60, label='FLUKA', zorder=5)

    axs[i].set_xscale('log')
    axs[i].set_yscale('log')
    axs[i].set_xlabel('Cooling time, h', fontsize=12)
    axs[i].set_ylabel(titles[i], fontsize=12)
    axs[i].grid(True, which='both', linestyle='--', alpha=0.4)
    axs[i].legend(loc='best', fontsize=10)

plt.tight_layout(pad=1.5)

# Сохраняем с именем "gp"
plt.savefig('gp.png', dpi=300, bbox_inches='tight')
plt.savefig('gp.pdf', format='pdf', dpi=300, bbox_inches='tight')
print("✅ Рисунок сохранён как gp.png и gp.pdf")
plt.show()
